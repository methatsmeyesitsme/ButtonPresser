<script>
const tokenEl = document.getElementById('token');
const ownerEl = document.getElementById('owner');
const repoEl = document.getElementById('repo');
const saveTokenEl = document.getElementById('saveToken');
const statusPill = document.getElementById('statusPill');
const checkBtn = document.getElementById('checkBtn');
const toggleBtn = document.getElementById('toggleBtn');
const msgEl = document.getElementById('msg');

let currentPrivate = null; // true / false / null

// Restore saved values (owner/repo always remembered for convenience; token only if opted in)
window.addEventListener('DOMContentLoaded', () => {
  const savedOwner = localStorage.getItem('rvt_owner');
  const savedRepo = localStorage.getItem('rvt_repo');
  const savedToken = localStorage.getItem('rvt_token');
  if (savedOwner) ownerEl.value = savedOwner;
  if (savedRepo) repoEl.value = savedRepo;
  if (savedToken) {
    tokenEl.value = savedToken;
    saveTokenEl.checked = true;
  }
});

function setMsg(text, type) {
  msgEl.textContent = text || '';
  msgEl.className = 'msg' + (type ? ' ' + type : '');
}

function setStatusPill(isPrivate) {
  if (isPrivate === null) {
    statusPill.textContent = 'Unknown';
    statusPill.className = 'pill unknown';
    toggleBtn.textContent = 'Check status first';
    toggleBtn.disabled = true;
    toggleBtn.className = '';
    return;
  }
  currentPrivate = isPrivate;
  if (isPrivate) {
    statusPill.textContent = 'Private';
    statusPill.className = 'pill private';
    toggleBtn.textContent = 'Make Public';
    toggleBtn.className = 'make-public';
  } else {
    statusPill.textContent = 'Public';
    statusPill.className = 'pill public';
    toggleBtn.textContent = 'Make Private';
    toggleBtn.className = 'make-private';
  }
  toggleBtn.disabled = false;
}

function getInputs() {
  const token = tokenEl.value.trim();
  const owner = ownerEl.value.trim();
  const repo = repoEl.value.trim();
  return { token, owner, repo };
}

function persistIfWanted() {
  const { token, owner, repo } = getInputs();
  if (owner) localStorage.setItem('rvt_owner', owner); else localStorage.removeItem('rvt_owner');
  if (repo) localStorage.setItem('rvt_repo', repo); else localStorage.removeItem('rvt_repo');
  if (saveTokenEl.checked && token) {
    localStorage.setItem('rvt_token', token);
  } else {
    localStorage.removeItem('rvt_token');
  }
}

async function apiRequest(method, body) {
  const { token, owner, repo } = getInputs();
  if (!token || !owner || !repo) {
    throw new Error('Please fill in token, owner, and repo name.');
  }
  const res = await fetch(`https://api.github.com/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}`, {
    method,
    headers: {
      'Authorization': `Bearer ${token}`,
      'Accept': 'application/vnd.github+json',
      'X-GitHub-Api-Version': '2022-11-28',
      ...(body ? { 'Content-Type': 'application/json' } : {})
    },
    body: body ? JSON.stringify(body) : undefined
  });

  if (!res.ok) {
    let detail = '';
    try {
      const errJson = await res.json();
      detail = errJson.message || '';
    } catch (e) {}
    if (res.status === 401) throw new Error('Bad credentials — check your token.');
    if (res.status === 403) throw new Error('Forbidden — token likely lacks Administration write permission on this repo.');
    if (res.status === 404) throw new Error('Repo not found — check owner/repo name, or token lacks access.');
    throw new Error(`GitHub API error (${res.status}): ${detail || 'unknown error'}`);
  }
  return res.json();
}

checkBtn.addEventListener('click', async () => {
  setMsg('');
  checkBtn.disabled = true;
  checkBtn.textContent = 'Checking...';
  try {
    persistIfWanted();
    const data = await apiRequest('GET');
    setStatusPill(!!data.private);
    setMsg(`Loaded ${data.full_name}.`, 'success');
  } catch (err) {
    setStatusPill(null);
    setMsg(err.message, 'error');
  } finally {
    checkBtn.disabled = false;
    checkBtn.textContent = 'Check current status';
  }
});

toggleBtn.addEventListener('click', async () => {
  if (currentPrivate === null) return;
  const target = !currentPrivate;
  const label = target ? 'private' : 'public';
  if (!confirm(`Make this repo ${label}? This takes effect immediately.`)) return;

  setMsg('');
  toggleBtn.disabled = true;
  const prevText = toggleBtn.textContent;
  toggleBtn.textContent = 'Updating...';
  try {
    persistIfWanted();
    const data = await apiRequest('PATCH', { private: target });
    setStatusPill(!!data.private);
    setMsg(`Repo is now ${data.private ? 'private' : 'public'}.`, 'success');
  } catch (err) {
    setMsg(err.message, 'error');
    toggleBtn.textContent = prevText;
    toggleBtn.disabled = false;
  }
});
</script>
